## 数据 Schema（字段约束，与 contracts/types.ts 一致）

```ts
{
  id: "op_<职业小写>_<两位序号>",   // 唯一，前缀必须 op_
  name: string,                      // 原创，不用明日方舟官方干员名
  profession: "vanguard"|"guard"|"defender"|"sniper"|"caster"|"medic"|"supporter"|"specialist",
  rarity: 3 | 4 | 5 | 6,
  placement: "melee"|"ranged",       // 先锋/近卫/重装/特种=melee；狙击/术师/医疗/辅助=ranged
  cost: number, blockCount: number,   // 近战 blockCount ≥ 1；高台干员 blockCount = 1
  hp/atk/def/res: number,
  attackInterval: number,             // 秒，1.0 ~ 2.4
  damageType: "physical"|"arts",
  range: [[dr,dc],...],              // 必含 [0,0]；近战通常 [[0,0]] 或 [[0,0],[0,1]]
  redeployCooldown: number,          // 秒，30 ~ 80
  dpBonusRate?: number,              // 仅先锋
  tags: string[],                    // 必含 "melee"/"ranged" 与职业名；远程另加 "ranged"
  description: string                // 一句话，剧团/猩红主题风格
}
```
